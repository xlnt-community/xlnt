52 // [[maybe_unused]] is supported in C++17 and newer, and C23 and newer
53 // C++17 needs an additional check as __has_cpp_attribute is only available with C++20 or higher (see above).
54 #if XLNT_HAS_CPP_ATTRIBUTE(maybe_unused) || XLNT_HAS_C_ATTRIBUTE(maybe_unused) || XLNT_HAS_CPP_VERSION(XLNT_CPP_17)
55  #define XLNT_UNUSED [[maybe_unused]]
56 // [[gnu::unused]] is supported in GCC and Clang since C++11. See:
57 // -> https://gcc.gnu.org/onlinedocs/gcc/Common-Attributes.html#index-unused
58 // -> https://clang.llvm.org/docs/LanguageExtensions.html#non-standard-c-11-attributes
59 //
60 // Note: [[...]] syntax is only supported by C++11 and newer, and C23 and newer.
61 #elif XLNT_HAS_CPP_ATTRIBUTE(gnu::unused) || XLNT_HAS_C_ATTRIBUTE(gnu::unused)
62  #define XLNT_UNUSED [[gnu::unused]]
63 // __attribute__((unused)) is supported in GCC and Clang. Can be useful with older compilers
64 // and in C code when compiled with a C version older than C23.
65 #elif XLNT_HAS_ATTRIBUTE(unused)
66  #define XLNT_UNUSED __attribute__((unused))
67 #else
68  #define XLNT_UNUSED
69 #endif
